    function array:scaleVec2(ScaleVec2:vector2){
        local ARRSC=array()
        foreach( N:number,Vec2:vector2=This){
          ARRSC[N,vector2]=Vec2*ScaleVec2
        }
        return ARRSC
    }
    
    function array:translVec2(TRVec2:vector2){
        local ARRSC=array()
        foreach(N:number,Vec2:vector2=This){
          ARRSC[N,vector2]=Vec2+TRVec2
        }
        return ARRSC
    }
    
        
    function array:rotVec2(Rot:number){
        local ARRSC=array()
        foreach(N:number,Vec2:vector2=This){
          ARRSC[N,vector2]=Vec2:rotate(Rot)
        }
        return ARRSC
    }
    
    function table:getPoly(Index:number){
