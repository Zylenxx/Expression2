    function array:scaleVec2(ScaleVec2:vector2){
        local ARRSC=array()
        foreach( N:number,Vec2:vector2=This){
          ARRSC[N,vector2]=Vec2*ScaleVec2
        }
        return ARRSC
    }
    
    function array:translVec2(TRVec2:vector2){
        local ARRSC=array()
        foreach(N:number,Vec2:vector2=This){
          ARRSC[N,vector2]=Vec2+TRVec2
        }
        return ARRSC
    }
    
        
    function array:rotVec2(Rot:number){
        local ARRSC=array()
        foreach(N:number,Vec2:vector2=This){
          ARRSC[N,vector2]=Vec2:rotate(Rot)
        }
        return ARRSC
    }
    
    function table:getPoly(Index:number){
        local Poly=array()
        foreach(N:number,ID:number=This[Index,array]){

        }
        return Poly
    }

    function wirelink:getNearestPoly(Pos:vector2){
        local C = TABL1:egpNumObjects()
        local Dst = array()
        for(I=1,C){
            local ID_OBJ_DIST=99999
                    ID_OBJ_DIST=This:egpPos(I):distance(Pos)
            Dst[I,number]=ID_OBJ_DIST
        }
        return Dst:minIndex()
    }
