@persist UIX UIY  R:ranger Sampled:vector DBG:string AOSZ REFLMAX

if(first()){
    holoCreate(1)
    holoPos(1,entity():pos()+vec(12.5,1.95,13.40):rotate(entity():angles()))
    holoModel(1,"models/hunter/misc/sphere025x025.mdl")
    holoScale(1,vec(0.05))
    holoMaterial(1,"models/debug/debugwhite")
    holoAlpha(1,64)
    holoDisableShading(1,1)
    holoParent(1,entity())
    
    W           = 1048572 # index for digi width
    H           = 1048573 # index for digi height
    N           = 1048574 # index for clear flag
    CLK         = 1048575 # clock (can be pulled high)
    COL         = 1048569 # index for color mode
    ZOOM        =    2000 # FOV (higher => higher FOV, not in degrees!)
    SAMPLES     =      24 # samples per pixel
    SAMPLEHEIGHT=    1200 # length of ray before stopping
    PERFLIMIT   =       1 # preferably do not change
    AOSZ        =     100 # length of ambient occlusion in HU
    CPULIM      =    3000 # default cpu limit state (adjustable with input ADD and REM at 100us intervals)
    REFLMAX     =       0 # maximum reflections on materials flagged either glass or metal (within shader)
    PTR         =       0
    PASSES      =       1 # how many times to go through before pausing. 0 sets to infinite.
    rangerPersist(1)
    rangerFilter(Ply)     # filters the entity set as the origin to avoid selfintersection
    if(!SETSZ){SETSZ=32}
    DS[W]=(SETSZ)*3/4
    DS[H]=(SETSZ)*4/4
    DS[N]=0
    DS[COL]=3
    DS[CLK]=1
    SCRP=-1
    function hasString(Input:string,Match:string){return Input:matchFirst(Match)==Match}
    

    function vector vector:doReflect(Hitnormal:vector){   
        local VN = Hitnormal:normalized()
        local RFV = vec()
        local ANG = 180
        local C = cos(ANG/2)
        local S = sin(ANG/2)
        local QConstr = quat(C,S*VN:x(),S*VN:y(),S*VN:z())
        RFV = vec(QConstr*This*conj(QConstr))
        return RFV
    }
     
    function doGlossyShader(MatType:string,FirstDir:vector,FirstHit:vector,R:ranger){
        local ColOut = vec(1)
        local WhereExit = rangerOffset(FirstHit+FirstDir,FirstHit+FirstDir*SAMPLEHEIGHT)
        local ExtraPass = rangerOffset(WhereExit:positionLeftSolid()+FirstDir*4,WhereExit:positionLeftSolid()+FirstDir*SAMPLEHEIGHT)
        if( ( (MatType=="glass") +(MatType=="metal") ) * REFLMAX ){
            local Roughn = (MatType=="metal")*0.25
            local R_REFL = FirstHit
            local INRANGER = R  
            local Dir = FirstDir
             for(_=1,REFLMAX){
             R_REFL=Dir:doReflect(R:hitNormal())
             INRANGER = rangerOffset(R:pos(),R:pos()+9999*(R_REFL+(Roughn*vec(random()-0.5,random()-0.5,random()-0.5)):normalized()))
             }
            local MPASS = 0.5+0.5*vec(INRANGER:hitNormal():dot(sunDirection()))
            local GTPASS=(0.5+0.5*vec(ExtraPass:hitNormal():dot(sunDirection())))
            if(MatType=="metal"){
            ColOut=MPASS
            }else{
            ColOut=max(vec(1),MPASS*0.3+0.9*GTPASS)
            }
        }
        return ColOut
    }
    
    # colormap, will get expanded upon later
    function cmap(E:entity){
     local C = vec(0.5)   
      if(E:getClass()=="player"){C=vec(0,1,0)} 
      if(E:getClass():sub(1,10)=="gmod_wire_"){C=vec(0.2,0.5,1)}
      if(E:getClass()=="prop_physics"){C=vec(1,0,0)}
      if(E:getClass()=="prop_ragdoll"){C=vec(0.7,0.7,1)}
      if(E:getClass()=="mining_rock"){C=vec(1,0.7,0.7)}
      if(E:getClass():sub(1,4)=="npc_"){C=vec(0,1,0)}
     return C   
    }
    
    function matCluster(S:string){
     local DIRS=S:explode("/")
     local Last=DIRS:count()
     local AllWeCareAbout = DIRS[Last,string]:right(2):toNumber(32)
     local COLM = vec(1)
     COLM = hsv2rgb((AllWeCareAbout*360/32+45)%360,0.8,1)*vec(1/255)
     return COLM   
    }
    function matInfer(S:string){
     local Yourmom =0
     local DIRS=S:explode("/")
     local ActualMatName=DIRS[DIRS:count(),string]:lower() 
     DBG = ActualMatName
     local COLM = vec(0.5,0.25,0)
      if (hasString(ActualMatName,"glass")       ){COLM = vec(0.9,0.9,1.0)}
      if (hasString(ActualMatName,"grass")       ){COLM = vec(0.2,1.0,1.0)}
      if (hasString(ActualMatName,"red")         ){COLM = vec(1.0,0.0,1.0)}
      if (hasString(ActualMatName,"green")       ){COLM = vec(0.0,1.0,0.0)}
      if (hasString(ActualMatName,"blue")        ){COLM = vec(0.0,0.0,1.0)}
      if (hasString(ActualMatName,"**empty**")   ){COLM = vec(0.2,0.2,0.2)}
      if (hasString(ActualMatName,"**studio**")  ){COLM = vec(0.0,0.9,1.0)}
      if (hasString(ActualMatName,"concrete")    ){COLM = vec(0.6)}
      if (hasString(ActualMatName,"metal")       ){COLM = vec(0.4)}
      if (hasString(ActualMatName,"wood")        ){COLM = vec(0.3,0.2,0.1)}
      if (hasString(ActualMatName,"displacement")){COLM = vec(1.0,0.5,1.0)}
     return COLM
    }
    
    
    # crude AO. requires multisampling to look good
    function getAO(P09852:vector,R18518){
        local AOContainer=0
        local CAO = vec(0)
            AOContainer+=rangerOffset(P09852,P09852+vec(0,0,R18518):rotate(ang(randint(360),randint(360),randint(360)))):hit()
            AOContainer+=rangerOffset(P09852,P09852+vec(0,0,R18518):rotate(ang(randint(360),randint(360),randint(360)))):hit()
            AOContainer+=rangerOffset(P09852,P09852+vec(0,0,R18518):rotate(ang(randint(360),randint(360),randint(360)))):hit()
            AOContainer+=rangerOffset(P09852,P09852+vec(0,0,R18518):rotate(ang(randint(360),randint(360),randint(360)))):hit()    
            AOContainer+=rangerOffset(P09852,P09852+vec(0,0,R18518):rotate(ang(randint(360),randint(360),randint(360)))):hit()
            AOContainer+=rangerOffset(P09852,P09852+vec(0,0,R18518):rotate(ang(randint(360),randint(360),randint(360)))):hit()
            AOContainer+=rangerOffset(P09852,P09852+vec(0,0,R18518):rotate(ang(randint(360),randint(360),randint(360)))):hit()
            AOContainer+=rangerOffset(P09852,P09852+vec(0,0,R18518):rotate(ang(randint(360),randint(360),randint(360)))):hit()
        CAO = vec(AOContainer/8)
        return CAO 
    }
}

setName("Raytracing: "+toString(DS[H])+"x"+toString(DS[W])+"\n@"+toString(SAMPLES)+" Samples/px,"+toString(8*SAMPLES)+" AO samples/px\nRT2 by Zylenxx (@zylenxx)")


event input(InputName:string) {
    if(changed(ADD)&&ADD){ADDCPUBIAS+=100}
    if(changed(REM)&&REM){ADDCPUBIAS-=100}
}

event tick(){
    PIXELRATEMEDIAN=max(1,(PIXELRATEMEDIAN*((PTR-1)/PTR)+(1/PTR)*PIXELRATE))
    PixelsToRender = DS[W]*DS[H]
    EstimateInSeconds = max(1,floor((PixelsToRender-PTR)/PIXELRATEMEDIAN))
    OP=ops()
    CT=cpuUsage()
    
    if(changed(floor(curtime()))){
        PIXELRATE=CURPX-PREVPX
        PREVPX=CURPX
        OldState=table(P,A,Ply:id())
    }
    
    holoColor(1,min(1,max(0,CPUTIME-ADDCPUBIAS-CPULIM)/1000)*vec(255,192,0))
    CPUTIME = cpuUsage()*1000000
    PERFLIMIT=max(2,(PERFLIMIT+((cpuUsage()*1000000)<(CPULIM+ADDCPUBIAS))-((cpuUsage()*1000000)>(CPULIM+ADDCPUBIAS))))
    for(_=1,PERFLIMIT/SAMPLES){ # scope branch 2
        if(maxquota()<500){continue}
        if(PASSES!=0 && SCRP>PASSES-1){continue}
        if(PTR==1){
            P=Ply:pos()
            A=Ply:angles()
            PIXELRATEMEDIAN=1
            SCRP+=1
        }
        
        X=PTR%DS[H]    
        Y=floor(PTR/DS[W])%DS[H]
        UIX=(X-(DS[W]/2))/DS[W]
        UIY=(Y-(DS[H]/2))/DS[H]
        local RNDX = 0
        local RNDY = 0
        for(N=1,SAMPLES){ #scope branch 3
            if(SAMPLES>1){ #scope branch 4
            local NewPat = (vec2(SAMPLES - floor(SAMPLES/3)*3, floor(SAMPLES/3)) - vec2(0, floor(SAMPLES/9)*3)) * 0.5 - vec2(0.5, 0.5) * (1 / floor(SAMPLES/12))
            RNDX = N>1*NewPat:x()*ZOOM/DS[W]
            RNDY = N>1*NewPat:y()*ZOOM/DS[H]
            # old ones:
            #RNDX = N>1*ZOOM/DS[W]*((randint(64)-32)/64)
            #RNDY = N>1*ZOOM/DS[H]*((randint(64)-32)/64)
            }                
            local RX=P+1/1000*vec(0,-UIX*ZOOM+RNDX,-UIY*ZOOM+RNDY):rotate(A)
            local RY=P+vec(SAMPLEHEIGHT,-UIX*ZOOM+RNDX,-UIY*ZOOM+RNDY):rotate(A)
            R=rangerOffset(
                RX,
                RY
                )
            local ALBEDO = min(1,2+1*R:hitNormal():dot(sunDirection()))
            
#[entity based]#
           # local CL     = cmap(R:entity())
#[Mat cluster]#
             #local CL     = matCluster(R:hitTexture()) 
#[Mat infer]#
             local CL     = matInfer(R:hitTexture()) 
#[Pure normals]#            
            #local CL     = vec(0.5)+vec(0.5)*R:hitNormal()
            local GLS = vec(1)
            local AOf = vec(1)
            if(R:hit()){
             GLS    = doGlossyShader(R:matType(),(RY-RX):normalized(),R:pos(),R)
             AOf    = min(vec(1),2*(vec(1)-getAO(R:pos()+R:hitNormal()*0.001,AOSZ))) #scope branch 5
            }
            local SH     = 1-0.5*rangerOffset(R:pos(),R:pos()-((randvec(vec(-0.01),vec(0.01)))+sunDirection())*SAMPLEHEIGHT):hit()
            local END    = R:hit()
            local SkyCol = vec(0.15,0.15,0.26)*(vec(1)-SH)
#[This is everything]#   Sampled      = (Sampled*((SAMPLES-1)/SAMPLES)) + ((1/SAMPLES)*(((ALBEDO+GLS)/2*AOf)*CL*SH*END+(SkyCol*(1-END))))
#[This is no AO]#     #  Sampled      = (Sampled*((SAMPLES-1)/SAMPLES)) + ((1/SAMPLES)*((ALBEDO+GLS)/2*CL*SH*END+(SkyCol*(1-END))))
#[This is just AO]#   #  Sampled=Sampled*((SAMPLES-1)/SAMPLES)+(1/SAMPLES)*(1-AOf)
        }
       # DS[DS[W]/2+(DS[W]*DS[H])/2]=rgb2digi(255,0,0,DS[COL])
        DS[PTR]=rgb2digi(255*Sampled,DS[COL])
        PTR+=1
        CURPX+=1
        PTR=PTR%(DS[W]*DS[H])   
    }
}
